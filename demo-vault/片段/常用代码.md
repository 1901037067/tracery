## 链接解析（Obsidian 规则）

```javascript
function resolveLink(target, notes, byName) {
  const clean = target.replace(/\.md$/i, '');
  if (notes.has(clean + '.md')) return clean + '.md';      // 1. 路径优先
  const hits = byName.get(clean.toLowerCase());
  return hits?.length === 1 ? hits[0] : null;              // 2. 文件名唯一匹配
}
```

## 光标按字符偏移保存/恢复

```javascript
// 重渲染前：把光标投影成纯文本偏移
function saveCaret(root) {
  const sel = getSelection();
  const range = sel.getRangeAt(0);
  const walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT);
  let count = 0, node;
  while ((node = walker.nextNode())) {
    if (node === range.startContainer) return count + range.startOffset;
    count += node.textContent.length;
  }
  return null;
}
```
